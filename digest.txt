import pino from "pino";
import { config } from "./index.js";

const baseLogger = pino({
  level: config.NODE_ENV === "development" ? "debug" : "info",
  ...(config.NODE_ENV === "development" && {
    transport: {
      target: "pino-pretty",
      options: {
        colorize: true,
        translateTime: "SYS:standard",
        ignore: "pid,hostname",
      },
    },
  }),
  base: {
    service: config.SERVICE_NAME,
  },
});

type LogMeta = Record<string, unknown> | Error | unknown;

type LogFn = {
  (msg: string, meta?: LogMeta): void;
  (meta: LogMeta, msg?: string): void;
};

const toMergingObject = (meta?: LogMeta): Record<string, unknown> => {
  if (meta === undefined || meta === null) {
    return {};
  }

  if (meta instanceof Error) {
    return {
      err: {
        message: meta.message,
        name: meta.name,
        stack: meta.stack,
      },
    };
  }

  if (typeof meta === "object") {
    return meta as Record<string, unknown>;
  }

  // primitive (string/number/boolean) â€“ wrap it
  return { value: meta };
};

const createLogFn = (level: "info" | "error" | "warn" | "debug"): LogFn => {
  return (msgOrMeta: string | LogMeta, maybeMetaOrMsg?: string | LogMeta) => {
    // Preferred: message first
    // logger.error("Error during graceful shutdown", error)
    // logger.info("Member added", { roomId })
    if (typeof msgOrMeta === "string") {
      const msg = msgOrMeta;
      const obj = toMergingObject(maybeMetaOrMsg);
      baseLogger[level](obj, msg);
      return;
    }

    // Also support: object/error first
    // logger.error(error, "Error during graceful shutdown")
    // logger.info({ roomId }, "Member added")
    const obj = toMergingObject(msgOrMeta);
    const msg = typeof maybeMetaOrMsg === "string" ? maybeMetaOrMsg : undefined;

    if (msg) {
      baseLogger[level](obj, msg);
    } else {
      baseLogger[level](obj);
    }
  };
};

export const logger = {
  info: createLogFn("info"),
  error: createLogFn("error"),
  warn: createLogFn("warn"),
  debug: createLogFn("debug"),
  child: baseLogger.child.bind(baseLogger),
};

import { Queue } from "bullmq";
import IORedis from "ioredis";
import { config } from "../config/index.js";
import { logger } from "../config/logger.js";

export const DELIVERY_QUEUE_NAME = "notification-deliveries";
export const DELIVERY_JOB_NAME = "process-delivery";

export type ProcessDeliveryJob = {
  deliveryId: string;
};

const connection = new IORedis(config.REDIS_URL, {
  maxRetriesPerRequest: null,
});

export const deliveryQueue = new Queue<ProcessDeliveryJob>(DELIVERY_QUEUE_NAME, {
  connection,
});

/**
 * Enqueue one delivery job.
 * Call this AFTER the DB transaction that created the Delivery row has committed.
 */

export async function enqueueDelivery(deliveryId: string): Promise<void>{
  await deliveryQueue.add(DELIVERY_JOB_NAME, { deliveryId }, {
    jobId: `delivery-${deliveryId}`,// idempotent: same deliveryId â†’ same job
    removeOnComplete: 1000,
    removeOnFail:5000
  })
  logger.debug("Enqueued delivery job", { deliveryId });
}

import cors from "cors";

const allowedOrigins = config.CLIENT_ORIGINS;

export const corsMiddleware = cors({
  origin: (origin, callback) => {
    // no Origin = curl / server-to-server
    if (!origin || allowedOrigins.includes(origin)) {
      callback(null, true);
      return;
    }
    callback(new Error(`Origin not allowed by CORS: ${origin}`));
  },
  credentials: true,
  methods: ["GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"],
  allowedHeaders: [
    "Content-Type",
    "Authorization",
    "X-API-Key",
    "X-Requested-With",
  ],
  exposedHeaders: ["Retry-After", "RateLimit-Limit", "RateLimit-Remaining"],
  optionsSuccessStatus: 204,
});

import type { Request, Response, NextFunction } from "express";
import { AppError } from "@notification/shared-utils";
import { logger } from "../config/logger.js";

export function errorMiddleware(
  err: Error,
  _req: Request,
  res: Response,
  _next: NextFunction,
): void {

  if (err instanceof AppError) {
    logger.warn({ err, code: err.code }, err.message);
    res.status(err.statusCode).json({
      success: false,
      error: {
        code: err.code,
        message: err.message,
        details: err.details,
      },
    });
    return;
  }

  logger.error({ err }, "Unhandled error");
  res.status(500).json({
    success: false,
    error: {
      code: "INTERNAL_ERROR",
      message:
        process.env.NODE_ENV === "production"
          ? "An unexpected error occurred"
          : err.message,
    },
  });
}

import type { NextFunction, Request, Response } from "express";
import { ZodType } from "zod";
import type { ParamsDictionary } from "express-serve-static-core";
import { BadRequestError } from "@notification/shared-utils";

type ValidationTarget = "body" | "params" | "query";

export const validate =
  <T>(schema: ZodType<T>, target: ValidationTarget = "body") =>
  (req: Request, res: Response, next: NextFunction): void => {
    const result = schema.safeParse(req[target]);
    if (!result.success) {
          const message = result.error.issues
                    .map(({ path, message }) => `${path.join(".")}:${message}`)
                    .join(",")
          return next(new BadRequestError(message));
    }
    switch (target) {
      case "body":
        req.body = result.data;
        break;

      case "params":
        req.params = result.data as ParamsDictionary;
        break;

        case "query":
                res.locals.validatedQuery = result.data;
                break;
    }

    next();
  };

import { Delivery, Notification, NotificationStatus } from "@notification/database";
import { NotificationInterface } from "./notification.interface.js";
import { CreateNotificationWriteData, NotificationWithDeliveries, UpdateDeliveryData } from "./notification.types.js";
import { prisma } from "../../infrastructure/prisma.js";

export class NotificationRepository implements NotificationInterface {
  async create(data: CreateNotificationWriteData): Promise<NotificationWithDeliveries> {
    const { channels, status, ...notificationData } = data
    return prisma.$transaction(async (tx) => {
      const notification = await tx.notification.create({
        data: {
          ...notificationData,
          status: status ?? "PENDING",
          deliveries: {
            create: channels.map((channel) => ({
              channel,
              status: "QUEUED",
              maxAttempts:3
            }))
          }
        },
        include:{deliveries:true}
        
      })
      return notification;
    })
  }

  async findByUserId(params: {
    userId: string;
    tenantId?: string;
    page: number;
    limit: number;
    unreadOnly?: boolean;
  }): Promise<{ data: Notification[]; total: number }> {
    const { userId, tenantId, page, limit, unreadOnly } = params;

    const where: any = {
      userId,
    };

    if (tenantId) {
      where.tenantId = tenantId;
    }

    if (unreadOnly) {
      where.readAt = null;
    }

    const [data, total] = await Promise.all([
      prisma.notification.findMany({
        where,
        orderBy: { createdAt: "desc" },
        skip: (page - 1) * limit,
        take: limit,
      }),
      prisma.notification.count({ where }),
    ]);

    return { data, total };
  }

// notification.service.ts

import {
  NotFoundError,
  ValidationError,
} from "@notification/shared-utils";
import type { DeliveryChannel } from "@notification/database";
import type { NotificationInterface } from "./notification.interface.js";
import type { CreateNotificationInput } from "./notification.schema.js";
import { enqueueDelivery } from "../../infrastructure/queue.js";
import { logger } from "../../config/logger.js";

const DEFAULT_CHANNELS: DeliveryChannel[] = ["IN_APP", "EMAIL"];

export class NotificationService {
  constructor(private readonly repo: NotificationInterface) {}

  async create(input: CreateNotificationInput) {
    // 1. Persist Notification + Deliveries (QUEUED) in one TX
    const notification = await this.repo.create({
      userId: input.userId,
      tenantId: input.tenantId ?? null,
      type: input.type,
      title: input.title,
      body: input.body,
      data: input.data ?? undefined,
      status: "PENDING",
      channels: DEFAULT_CHANNELS,
    });

    // 2. Enqueue one job per delivery (after TX committed)
    for (const delivery of notification.deliveries) {
      await enqueueDelivery(delivery.id);
    }

    logger.info("Notification created and jobs enqueued", {
      notificationId: notification.id,
      deliveryIds: notification.deliveries.map((d) => d.id),
    });

    // Status is still PENDING; worker will move it to DELIVERED / PARTIAL / FAILED
    return notification;
  }

  async getUserNotifications(params: {
    userId: string;
    tenantId?: string;
    page: number;
    limit: number;
    unreadOnly?: boolean;
  }) {
    const { userId, tenantId, page, limit, unreadOnly } = params;
    if (!userId) throw new ValidationError("userId is required");

    const { data, total } = await this.repo.findByUserId({
      userId,
      tenantId,
      page,
      limit,
      unreadOnly,
    });

    return {
      data,
      meta: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit) || 0,
      },
    };
  }

  async findNotificationById(id: string) {
    if (!id) throw new ValidationError("Notification id is required");
    const notification = await this.repo.findById(id);
    if (!notification) throw new NotFoundError("Notification", id);
    return notification;
  }

  async findNotificationByIdWithDeliveries(id: string) {
    if (!id) throw new ValidationError("Notification id is required");
    const notification = await this.repo.findByIdWithDeliveries(id);
    if (!notification) throw new NotFoundError("Notification", id);
    return notification;
  }

  async markAsRead(id: string) {
    if (!id) throw new ValidationError("Notification id is required");
    const notification = await this.repo.findById(id);
    if (!notification) throw new NotFoundError("Notification", id);
    if (notification.readAt) return notification;
    return this.repo.markAsRead(id);
  }
}

import { Worker } from "bullmq";
import { logger } from "./config/logger.js";
import { prisma } from "./infrastructure/prisma.js";
import {
  createRedisConnection,
  DELIVERY_QUEUE_NAME,
  type ProcessDeliveryJob,
} from "./infrastructure/queue.js";
import { createDeliveryProcessor } from "./processors/delivery.processor.js";
import { InAppAdapter } from "./adapters/in-app.adapter.js";
import { EmailAdapter } from "./adapters/email.adapter.js";
import type { DeliveryChannel } from "@notification/database";
import type { ChannelAdapter } from "./adapters/channel-adapter.interface.js";

async function main() {
  const connection = createRedisConnection();

  const adapters = new Map<DeliveryChannel, ChannelAdapter>([
    ["IN_APP", new InAppAdapter()],
    ["EMAIL", new EmailAdapter()],
  ]);

  const processor = createDeliveryProcessor(adapters);

  const worker = new Worker<ProcessDeliveryJob>(
    DELIVERY_QUEUE_NAME,
    processor,
    {
      connection,
      concurrency: 5,
    },
  );

  worker.on("completed", (job) => {
    logger.info("Job completed", { jobId: job.id, deliveryId: job.data.deliveryId });
  });

  worker.on("failed", (job, err) => {
    logger.error("Job failed", {
      jobId: job?.id,
      deliveryId: job?.data.deliveryId,
      error: err.message,
    });
  });

  logger.info(`notification-worker started (queue=${DELIVERY_QUEUE_NAME})`);

  const shutdown = async (signal: string) => {
    logger.info(`Shutting down worker (${signal})`);
    await worker.close();
    await connection.quit();
    await prisma.$disconnect();
    process.exit(0);
  };

  process.on("SIGINT", () => void shutdown("SIGINT"));
  process.on("SIGTERM", () => void shutdown("SIGTERM"));
}

main().catch((err) => {
  logger.error(err, "Failed to start worker");
  process.exit(1);
});

const baseLogger = pino({
  level: config.NODE_ENV === "development" ? "debug" : "info",
  ...(config.NODE_ENV === "development" && {
    transport: {
      target: "pino-pretty",
      options: {
        colorize: true,
        translateTime: "SYS:standard",
        ignore: "pid,hostname",
      },
    },
  }),
  base: {
    service: config.SERVICE_NAME,
  },
});

type LogMeta = Record<string, unknown> | Error | unknown;

type LogFn = {
  (msg: string, meta?: LogMeta): void;
  (meta: LogMeta, msg?: string): void;
};

const toMergingObject = (meta?: LogMeta): Record<string, unknown> => {
  if (meta === undefined || meta === null) {
    return {};
  }

  if (meta instanceof Error) {
    return {
      err: {
        message: meta.message,
        name: meta.name,
        stack: meta.stack,
      },
    };
  }

  if (typeof meta === "object") {
    return meta as Record<string, unknown>;
  }

  // primitive (string/number/boolean) â€“ wrap it
  return { value: meta };
};

const createLogFn = (level: "info" | "error" | "warn" | "debug"): LogFn => {
  return (msgOrMeta: string | LogMeta, maybeMetaOrMsg?: string | LogMeta) => {
    // Preferred: message first
    // logger.error("Error during graceful shutdown", error)
    // logger.info("Member added", { roomId })
    if (typeof msgOrMeta === "string") {
      const msg = msgOrMeta;
      const obj = toMergingObject(maybeMetaOrMsg);
      baseLogger[level](obj, msg);
      return;
    }

    // Also support: object/error first
    // logger.error(error, "Error during graceful shutdown")
    // logger.info({ roomId }, "Member added")
    const obj = toMergingObject(msgOrMeta);
    const msg = typeof maybeMetaOrMsg === "string" ? maybeMetaOrMsg : undefined;

    if (msg) {
      baseLogger[level](obj, msg);
    } else {
      baseLogger[level](obj);
    }
  };
};

export const logger = {
  info: createLogFn("info"),
  error: createLogFn("error"),
  warn: createLogFn("warn"),
  debug: createLogFn("debug"),
  child: baseLogger.child.bind(baseLogger),
};

import type { Job } from "bullmq";
import type { DeliveryChannel, NotificationStatus } from "@notification/database";
import { prisma } from "../infrastructure/prisma.js";
import type { ProcessDeliveryJob } from "../infrastructure/queue.js";
import type { ChannelAdapter } from "../adapters/channel-adapter.interface.js";
import { logger } from "../config/logger.js";

export function createDeliveryProcessor(
  adapters: Map<DeliveryChannel, ChannelAdapter>,
) {
  return async function processDelivery(job: Job<ProcessDeliveryJob>) {
    const { deliveryId } = job.data;

    logger.info("Processing delivery job", {
      jobId: job.id,
      deliveryId,
    });

    const delivery = await prisma.delivery.findUnique({
      where: { id: deliveryId },
      include: { notification: true },
    });

    if (!delivery) {
      logger.warn("Delivery not found, skipping", { deliveryId });
      return;
    }

    // Idempotency: only process QUEUED
    if (delivery.status !== "QUEUED") {
      logger.info("Delivery already processed, skipping", {
        deliveryId,
        status: delivery.status,
      });
      return;
    }

    // CAS: QUEUED â†’ SENDING, attempts++
    const cas = await prisma.delivery.updateMany({
      where: { id: deliveryId, status: "QUEUED" },
      data: {
        status: "SENDING",
        attempts: { increment: 1 },
      },
    });

    if (cas.count === 0) {
      logger.info("Lost CAS race, skipping", { deliveryId });
      return;
    }

    const adapter = adapters.get(delivery.channel);
    if (!adapter) {
      await markFailed(deliveryId, `No adapter for channel ${delivery.channel}`);
      await rollUpNotificationStatus(delivery.notificationId);
      return;
    }

    try {
      const result = await adapter.send({
        notificationId: delivery.notificationId,
        userId: delivery.notification.userId,
        title: delivery.notification.title,
        body: delivery.notification.body,
        data: delivery.notification.data ?? undefined,
      });

      await prisma.delivery.update({
        where: { id: deliveryId },
        data: {
          status: "SENT",
          sentAt: new Date(),
          providerMessageId: result.providerMessageId ?? null,
          lastError: null,
        },
      });

      logger.info("Delivery sent", {
        deliveryId,
        channel: delivery.channel,
        providerMessageId: result.providerMessageId,
      });
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      logger.error("Delivery send failed", {
        deliveryId,
        channel: delivery.channel,
        error: message,
      });
      await markFailed(deliveryId, message);
    }

    await rollUpNotificationStatus(delivery.notificationId);
  };
}

async function markFailed(deliveryId: string, lastError: string) {
  await prisma.delivery.update({
    where: { id: deliveryId },
    data: {
      status: "FAILED",
      lastError,
    },
  });
}

async function rollUpNotificationStatus(notificationId: string) {
  const deliveries = await prisma.delivery.findMany({
    where: { notificationId },
  });

  if (deliveries.length === 0) {
    await prisma.notification.update({
      where: { id: notificationId },
      data: { status: "FAILED" },
    });
    return;
  }

  const allSent = deliveries.every((d) => d.status === "SENT");
  const allFailed = deliveries.every(
    (d) => d.status === "FAILED" || d.status === "DEAD",
  );
  const anySent = deliveries.some((d) => d.status === "SENT");
  const anyPending = deliveries.some(
    (d) => d.status === "QUEUED" || d.status === "SENDING",
  );

  let status: NotificationStatus;
  if (anyPending) status = "PENDING";
  else if (allSent) status = "DELIVERED";
  else if (allFailed) status = "FAILED";
  else if (anySent) status = "PARTIAL";
  else status = "FAILED";

  await prisma.notification.update({
    where: { id: notificationId },
    data: { status },
  });

  logger.info("Notification status rolled up", {
    notificationId,
    status,
  });
}

import { PrismaPg } from "@prisma/adapter-pg";
import { PrismaClient } from "./generated/prisma/client.js";
import pg from "pg";

export type CreatePrismaOptions = {
  connectionString: string;
  log?: Array<"error" | "warn" | "info" | "query">;
};

/**
 * Factory so each app gets its own pool but the same generated client types.
 */
export function createPrismaClient(options: CreatePrismaOptions): PrismaClient {
  const pool = new pg.Pool({ connectionString: options.connectionString });
  const adapter = new PrismaPg(pool);

  return new PrismaClient({
    adapter,
    log: options.log ?? ["error"],
  });
}

export { PrismaClient };
export type {
  NotificationStatus,
  Prisma,
  Notification,
  Delivery,
  DeliveryStatus,
  DeliveryChannel
} from "./generated/prisma/client.js";

import type { Request, Response, NextFunction, RequestHandler } from "express";

export function asyncHandler(
  fn: (req: Request, res: Response, next: NextFunction) => Promise<unknown>,
): RequestHandler {
  return (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next);
  };
}

export function generateId(prefix = ""): string {
  const id = crypto.randomUUID().replace(/-/g, "");
  return prefix ? `${prefix}_${id}` : id;
}

export function calculateBackoffMs(attempt: number, baseMs = 1000, maxMs = 60_000): number {
  const exp = Math.min(maxMs, baseMs * 2 ** (attempt - 1));
  // full jitter
  return Math.floor(Math.random() * exp);
}
